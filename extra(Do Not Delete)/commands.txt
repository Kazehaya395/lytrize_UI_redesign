Cleanup - 
Windows - 
1. Delete specific top-level build and cache directories
Remove-Item -Recurse -Force build, dist, .build, .pytest_cache, .mypy_cache, .ruff_cache, .eggs, htmlcov -EA SilentlyContinue
# 2. Delete the specific build_hooks pycache folder
Get-ChildItem -Path ".\" -Filter "__pycache__" -Directory -Recurse | Remove-Item -Recurse -Force -ErrorAction SilentlyContinue
# 3. Find and delete all other __pycache__ directories in source/tools
Get-ChildItem packtoworkeu, tools -Recurse -Directory -Filter __pycache__ -EA SilentlyContinue | Remove-Item -Recurse -Force -EA SilentlyContinue
# 4. Find and delete leftover compiled files and coverage data
Get-ChildItem packtoworkeu, tools -Recurse -File -Include *.pyc, *.pyo, .coverage -EA SilentlyContinue | Remove-Item -Force -EA SilentlyContinue

Linux -
find . -type d \( -name "__pycache__" -o -name ".pytest_cache" -o -name ".mypy_cache" -o -name ".ruff_cache" -o -name "build" -o -name ".build" -o -name "dist" -o -name "*.egg-info" -o -name ".eggs" -o -name "htmlcov" \) -not -path "./venv/*" -prune -exec rm -rf {} + 2>/dev/null
find . -type f \( -name "*.pyc" -o -name "*.pyo" -o -name ".coverage" \) -not -path "./venv/*" -delete 2>/dev/null
rm -rf ~/.packtoworkeu
